    \question به سوالات زیر پاسخ دهید.

    \begin{parts}

        \part تفاوت اصلی فرآیندهای مستقل (Independent) و همکار (Cooperating) چیست؟
        \part سه چالش اصلی در ارتباط فرآیندهای همکار را نام ببرید.
        \part شرط انحصار متقابل \lr{(Mutual Exclusion)} در دسترسی به بافر مشترک به چه معناست؟
        \part بخش باقی‌مانده \lr{(Remainder Section)} در ساختار حل مسئله ناحیه بحرانی شامل چه کدهایی است؟
        \part چگونه اجرای هم‌روند دستورات ساده‌ای مانند \texttt{count++} و \texttt{count--} در سطح زبان ماشین می‌تواند منجر به ناهماهنگی داده‌ها شود؟

    \end{parts}

    \begin{solution}
        \begin{parts}
            \part فرآیندهای مستقل داده مشترک ندارند و بر یکدیگر تاثیر نمی‌گذارند، اما فرآیندهای همکار می‌توانند از فرآیندهای دیگر تاثیر بپذیرند یا بر اجرای آن‌ها اثر بگذارند.
            \part این چالش‌ها شامل همگام‌سازی (Synchronization)، تبادل اطلاعات (Communication) و رقابت (Race) هستند.
            \part بدین معناست که در هر لحظه تنها یک فرآیند باید توانایی دسترسی به بافر مشترک و تغییر آن را داشته باشد.
            \part به تمامی کدهای فرآیند که خارج از سه بخش اصلی ورود (Entry)، خروج (Exit) و ناحیه بحرانی \lr{(Critical Section)} قرار دارند، بخش باقی‌مانده اطلاق می‌شود.
            \part این دستورات در سطح ماشین به چندین دستورالعمل مجزا (مانند بارگذاری در رجیستر، انجام محاسبات و ذخیره مجدد) تقسیم می‌شوند. در صورت وقوع تعویض متن (Context Switch) در میان این مراحل، ممکن است مقدار نهایی ذخیره شده در حافظه بر اساس مقادیر قدیمی رجیسترها بازنویسی شود و نتیجه‌ای ناصحیح (مثلاً ۴ یا ۶ به جای ۵) حاصل گردد.
        \end{parts}
    \end{solution}

    \question
    \begin{parts}
        \part سه نیازمندی اساسی برای یک راه حل معتبر در مسئله ناحیه بحرانی را نام ببرید.
        \part چرا در تلاش اول دکر (Decker)، ویژگی «پیشرفت» نقض می‌شود؟
        \part مشکل اصلی تلاش دوم دکر که منجر به نقض انحصار متقابل می‌شود چیست؟
        \part تفاوت مفهومی بن‌بست (Deadlock) و قفل زنده (Livelock) در چیست؟
        \part رابطه منطقی بین پدیده گرسنگی (Starvation)، بن‌بست (Deadlock) و ویژگی انتظار محدود را تحلیل کنید.
    \end{parts}

    \begin{solution}
        \begin{parts}
            \part این سه نیازمندی عبارتند از: انحصار متقابل \lr{(Mutual Exclusion)}، پیشرفت \lr{(Progress)} و انتظار محدود \lr{(Bounded Waiting)}.
            \part چون ورود به ناحیه بحرانی به صورت نوبتی اجباری شده است؛ اگر فرآیندی که نوبتش است در ناحیه غیربحرانی بماند، فرآیند دیگر حتی در صورت نیاز، اجازه ورود به ناحیه بحرانی را نخواهد داشت.
            \part مشکل در اینجاست که بررسی وضعیت پرچم (flag) و تغییر آن به صورت اتمیک انجام نمی‌شود؛ در نتیجه ممکن است هر دو فرآیند هم‌زمان پرچم دیگری را \texttt{FALSE} مشاهده کرده و هر دو وارد ناحیه بحرانی شوند.
            \part در بن‌بست فرآیندها به طور ابدی متوقف شده و منتظر می‌مانند، اما در قفل زنده فرآیندها مدام تغییر وضعیت می‌دهند ولی به دلیل عدم هماهنگی در سرعت نسبی، عملاً پیشرفتی حاصل نمی‌شود.
            \part طبق منابع، اگر در سیستمی تضمین شود که گرسنگی رخ نمی‌دهد ($\neg Starvation$)، می‌توان نتیجه گرفت که بن‌بست و قفل زنده نیز هرگز رخ نخواهند داد. در واقع، عدم وقوع گرسنگی به معنای برقراری شرط انتظار محدود است که سلامت پروتکل را از نظر زنده بودن (Liveness) تضمین می‌کند.
        \end{parts}
    \end{solution}

    \question
    \begin{parts}
        \part مشکل اصلی «تلاش چهارم الگوریتم دکر» چیست و در تلاش پنجم چگونه برطرف شده است؟
        \part در الگوریتم پترسون (برای دو فرآیند)، متغیرهای flag و turn چه نقش متفاوتی ایفا می‌کنند؟
        \part چرا روش سخت‌افزاری «از کار انداختن وقفه» \lr{(Disable Interrupts)} برای سیستم‌های چندپردازنده‌ای مناسب نیست؟
        \part دستورالعمل TSL \lr{(Test and Set Lock)} چگونه کار می‌کند و کدام یک از سه شرط اصلی مسئله ناحیه بحرانی را ارضا نمی‌کند؟
    \end{parts}

    \begin{solution}
        \begin{parts}
            \part مشکل اصلی تلاش چهارم، وقوع قفل زنده (Livelock) است. در این سناریو، فرآیندها وارد یک سیکل معیوب از «ادب و تعارف» می‌شوند؛ هر دو پرچم (flag) خود را پایین می‌آورند تا به دیگری اجازه دهند و دوباره هم‌زمان آن را بالا می‌برند، در نتیجه هیچ‌کدام وارد ناحیه بحرانی نمی‌شوند. در تلاش پنجم، دکر متغیر turn (نوبت) را با flag ترکیب کرد. در این حالت، اگر هر دو فرآیند هم‌زمان درخواست ورود داشته باشند، تنها فرآیندی که نوبت (turn) با اوست پافشاری می‌کند و فرآیند دیگر وارد فاز انتظار می‌شود تا نوبت به او برسد. این کار هم‌زمانی انحصار متقابل، پیشرفت و انتظار محدود را تضمین می‌کند.
            \part در الگوریتم پترسون:
            \begin{itemize}
                \item متغیر flag نشان‌دهنده تمایل و آمادگی فرآیند برای ورود به ناحیه بحرانی است.
                \item متغیر turn تعیین‌کننده حق تقدم است. هر فرآیند پس از اعلام آمادگی، با تنظیم turn به مقدار شناسه فرآیند دیگر، با «ادب» نوبت را به دیگری تعارف می‌کند. اگر هر دو هم‌زمان درخواست دهند، فرآیندی وارد می‌شود که turn به نام او نیست (یا به عبارتی، فرآیندی که دیرتر turn را تغییر داده است، نوبت را به دیگری داده و خودش منتظر می‌ماند).
            \end{itemize}
            \part زیرا در سیستم‌های چندپردازنده‌ای، غیرفعال کردن وقفه تنها روی پردازنده‌ای که دستور را اجرا کرده اثر می‌گذارد. سایر فرآیندها که روی پردازنده‌های دیگر در حال اجرا هستند، همچنان می‌توانند به ناحیه بحرانی دسترسی داشته باشند و شرط انحصار متقابل نقض می‌شود. همچنین اگر فرآیندی در ناحیه بحرانی از کار بیفتد، کل سیستم (به دلیل غیرفعال ماندن وقفه‌ها) دچار اختلال می‌شود.
            \part دستور TSL به صورت اتمیک (غیرقابل تقسیم) محتویات کلمه حافظه (Lock) را می‌خواند و مقدار آن را به 1 تغییر می‌دهد. اگر مقدار خوانده شده 0 باشد، فرآیند وارد ناحیه بحرانی می‌شود؛ اگر 1 باشد، در حلقه انتظار مشغول می‌ماند. این روش شرط انتظار محدود \lr{(Bounded Waiting)} را ارضا نمی‌کند. زیرا اگر چند فرآیند منتظر باشند و فرآیند داخل ناحیه بحرانی خارج شود، انتخاب فرآیند بعدی تصادفی یا وابسته به سخت‌افزار است و ممکن است یک فرآیند دچار گرسنگی شده و تا ابد منتظر بماند.
        \end{parts}
    \end{solution}

    \question با مقایسه الگوریتم پترسون (راهکار نرم‌افزاری) و دستورالعمل TSL (راهکار سخت‌افزاری)، هر دو از مکانیسم «انتظار مشغول» \lr{(Busy Waiting)} استفاده می‌کنند. با این حال، چرا گفته می‌شود الگوریتم پترسون شرط «انتظار محدود» را برآورده می‌کند، اما TSL ممکن است باعث گرسنگی (Starvation) شود؟ (تحلیل کنید چه مکانیزمی در پترسون وجود دارد که در TSL غایب است).
    \begin{solution}
        اگرچه هر دو روش از انتظار مشغول \lr{(while loop)} استفاده می‌کنند، تفاوت بنیادین در نحوه مدیریت نوبت است:

        \begin{itemize}
            \item در الگوریتم پترسون:
                  یک متغیر صریح به نام
                  turn وجود دارد که ترتیب ورود را کنترل می‌کند. وقتی فرآیندی از ناحیه بحرانی خارج می‌شود یا قصد ورود دارد، وضعیت متغیرهای flag و turn به گونه‌ای تغییر می‌کند که فرآیند دیگر که در حلقه انتظار گیر افتاده است، شرط حلقه‌اش (while) نقض شده و حتماً اجازه ورود پیدا می‌کند. این ساختار منطقی تضمین می‌کند که اگر فرآیند P0 منتظر باشد، P1 نمی‌تواند چندین بار پشت سر هم وارد ناحیه بحرانی شود و نوبت نهایتاً به P0 می‌رسد (جلوگیری از گرسنگی).

            \item در دستورالعمل TSL: هیچ متغیری برای حفظ «نوبت» یا «صف» وجود ندارد. تمام فرآیندهای منتظر، دائماً در حال تلاش برای اجرای دستور TSL هستند تا قفل (lock) را بگیرند. وقتی قفل آزاد می‌شود (صفر می‌شود)، اینکه کدام فرآیند زودتر دستور TSL را اجرا کند و قفل را دوباره یک کند، کاملاً وابسته به سرعت پردازنده و زمان‌بندی سخت‌افزاری است. ممکن است یک فرآیند سریع‌تر همیشه برنده شود و فرآیند دیگر دائماً در رقابت شکست بخورد و دچار گرسنگی شود.
        \end{itemize}
    \end{solution}

    \question
    قفل‌های \lr{mutex} که بر پایه «انتظار مشغول» (\lr{busy-waiting}) هستند و
    با نام \lr{spin-lock} شناخته می‌شوند،
    در چه شرایطی نسبت به قفل‌های مبتنی بر «مسدود کردن» (\lr{blocking}) ترجیح داده می‌شوند؟
    \\
    \begin{solution}

        قفل‌های \lr{spin-lock} زمانی گزینه بهتری هستند که \textbf{زمان انتظار برای قفل بسیار کوتاه باشد}. دلیل این امر آن است که مسدود کردن یک فرآیند و بیدار کردن مجدد آن مستلزم انجام \textbf{دو تعویض متن} (یکی برای به خواب رفتن و دیگری برای بیدار شدن) است که هزینه زمانی و پردازشی بالایی دارد. اگر زمان لازم برای چرخش (\lr{spinning}) در انتظار قفل، کمتر از زمان مورد نیاز برای تعویض متن باشد، استفاده از \lr{spin-lock} به صرفه‌تر خواهد بود. این قفل‌ها به‌ویژه در \textbf{سیستم‌های چند هسته‌ای} مفید هستند، زیرا یک هسته می‌تواند در حال اجرای بخش بحرانی باشد و هسته دیگر بدون نیاز به تعویض متن، برای مدتی کوتاه منتظر آزاد شدن آن بماند.
    \end{solution}

    \question فرض کنید در کد زیر
    \lr{p1} و \lr{p2}
    همروند باشند و سمافورهای x و y دو سمافور دودویی و هر دو مقدار اولیه یک را داشته باشند. همچنین
    فرض کنید دستور print در یک چرخه دستورالعمل اجرا شود.

    \begin{latin}
        \begin{center}
            \begin{lstlisting}[basicstyle=\ttfamily\small]
p2:                         p1:
----                        ----
wait(y)                     wait(x)
while true do               while true do
print "B"                   print "A"
signal(y)                   signal(x)
----                        ----
            \end{lstlisting}
        \end{center}
    \end{latin}

    آیا هر تلفیقی از A ها و B ها ممکن است در خروجی ظاهر شوند؟ توضیح دهید.
    \\
    \begin{solution}
        با توجه به اینکه این دو فرآیند از سمافور‌های متفاوتی استفاده می‌کنند هیچ کدام منتظر دیگری نمی‌ماند.
        با فرض اینکه این دو فرآیند همروند اجرا شوند، بنابراین در هر قسمتی از اجرا ممکن است اجرای آن‌ها متوقف شود
        و CPU به فرآیند دیگر اختصاص داده شود که این بدان معناست که
        خروجی این برنامه‌ها از الگوی خاصی پیروی نمی‌کند و هر حالتی ممکن است داشته باشد.
    \end{solution}

    \question
    ساختار monitor چه مشکلاتی را حل می‌کند که قفل‌های mutex و monitor با آنها دسته پنجه نرم می‌کنند؟
    این کار را به چه شکلی انجام می‌دهد؟
    \\
    \begin{solution}
        استفاده مستقیم از سمافورها مستعد \textbf{خطاهای منطقی و زمانی} است؛ برای مثال، جابجا نوشتن ترتیب \lr{wait} و \lr{signal}، فراموش کردن یکی از آن‌ها، یا استفاده مکرر از \lr{wait} روی یک سمافور که می‌تواند منجر به نقض انحصار متقابل یا ایجاد \textbf{بن‌بست (\lr{deadlock})} دائمی شود. \lr{monitor} این مشکل را با ارائه یک \textbf{نوع داده انتزاعی (\lr{ADT})} حل می‌کند. مانیتور متغیرهای مشترک و توابع دسترسی به آن‌ها را در یک بسته واحد کپسوله می‌کند و \textbf{تضمین می‌کند که در هر لحظه تنها یک فرآیند در داخل مانیتور فعال باشد}. با این مکانیزم، برنامه‌نویس نیازی به کدنویسی صریح برای مدیریت قفل‌ها ندارد و احتمال خطاهای انسانی به شدت کاهش می‌یابد.
    \end{solution}

    \question
    \begin{parts}
        \part نحوه همگام‌سازی در مسئله تولیدکننده-مصرف‌کننده را با استفاده از مانیتور شرح دهید.
        \part یک نمونه مانیتور برای حل مسئله «غذا خوردن فیلسوف‌ها» بنویسید که از بن‌بست جلوگیری کند.
    \end{parts}

        \begin{parts}
            \part        در راهکار مانیتور برای تولیدکننده-مصرف‌کننده، از دو متغیر شرطی (\lr{condition variables}) به نام‌های \lr{full}
            و \lr{empty} استفاده می‌شود. اگر بافر پر باشد، تولیدکننده با دستور \lr{full.wait()} معلق می‌شود.
            وقتی مصرف‌کننده آیتمی را برمی‌دارد، با \lr{full.signal()} به تولیدکننده اطلاع می‌دهد.
            به همین ترتیب، اگر بافر خالی باشد، مصرف‌کننده روی \lr{empty.wait()} منتظر می‌ماند تا تولیدکننده پس از درج آیتم جدید، \lr{empty.signal()} را فراخوانی کند.

            \part نمونه مانیتور:
            \begin{latin}
                \begin{lstlisting}
monitor DiningPhilosophers { 
    enum {THINKING, HUNGRY, EATING} state; 
    condition self;

    function pickup(int i) {
        state[i] = HUNGRY;
        test(i);
        if (state[i] != EATING) self[i].wait();
    }

    function putdown(int i) {
        state[i] = THINKING;
        test((i + 4) % 5); // Check left neighbor
        test((i + 1) % 5); // Check right neighbor
    }

    function test(int i) {
        if ((state[(i + 4) % 5] != EATING)
            && (state[i] == HUNGRY)
            && (state[(i + 1) % 5] != EATING))
        {
            state[i] = EATING;
            self[i].signal();
        }
    }

    initialization_code() {
        for (int i = 0; i < 5; i++) state[i] = THINKING;
    }
}
    \end{lstlisting}
            \end{latin}
        \end{parts}
    \end{solution}

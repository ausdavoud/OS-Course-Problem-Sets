\begin{parts}
    
    \part صف حلقوی \lr{(Circular Queue)}.
    \part  آن‌ها باید توافق کنند که محدودیت سیستم عامل را که در حالت عادی مانع دسترسی یک فرآیند به فضای آدرس فرآیند دیگر می‌شود، حذف کنند.
    \part{۱۴.} فرآیندها باید ضمانت کنند که به طور همزمان روی مکان یکسانی نمی‌نویسند.
    \part{۱۵.} از طریق حافظه اشتراکی، گذرگاه سخت‌افزاری، یا شبکه.
    \part{۱۶.} ۱. امکان ایجاد صندوق پستی جدید. ۲. امکان ارسال و دریافت پیام از طریق صندوق پستی. ۳. امکان حذف صندوق پستی.
    \part{۱۷.} به این معنی است که اگر بخشی از برنامه بلاک شود، سایر بخش‌ها می‌توانند به اجرا ادامه دهند.
    \part{۱۸.} ۱. شناسه نخ \lr{(Thread ID)}، ۲. شمارنده برنامه \lr{(Program Counter)}، ۳. مجموعه ثبات \lr{(Register Set)}، ۴. پشته \lr{(Stack)}.
    \part{۱۹.} ۱. بخش کد، ۲. بخش داده، ۳. منابع سیستم‌عامل (مانند فایل‌های باز).
    \part{۲۰.} ۱. پاسخگویی سریع، ۲. اشتراک‌گذاری منابع، ۳. صرفه اقتصادی، ۴. مقیاس‌پذیری.
    \part{۲۱.} ۱. شناسایی وظایف، ۲. تعادل، ۳. جداسازی داده، ۴. وابستگی داده، ۵. تست و اشکال‌زدایی.
    \part{۲۲.} زیرا بلاک شدن یک نخ در سطح کاربر باعث بلاک شدن کل فرآیند نمی‌شود و سایر نخ‌ها می‌توانند به کار خود ادامه دهند.
    \part{۲۳.} ۱. سربار زمانی ایجاد نخ و حذف آن پس از اتمام کار. ۲. عدم وجود محدودیت برای تعداد نخ‌های فعال که می‌تواند باعث هدررفت منابع شود.
    \part{۲۴.} داده‌های ایستا معمولاً بین همه نخ‌ها مشترک هستند، اما داده‌های TLS برای هر نخ یکتا و اختصاصی هستند.

        \textbf{دلایل رخ‌دادن:}
        \begin{itemize}
            \item Mailbox ظرفیت محدود دارد و وقتی پر شود، تسک فرستنده بلوکه می‌شود و زمان پاسخ بالا می‌رود.
            \item ارسال و دریافت پیام شامل کپی‌کردن داده و \lr{Context Switch} است که در بار سنگین باعث افزایش تأخیر می‌شود.
            \item اگر Consumer اولویت پایین داشته باشد، خالی‌شدن Mailbox دیر انجام می‌شود و عملاً \lr{Priority Inversion} رخ می‌دهد.
        \end{itemize}

        \textbf{راه‌حل‌ها:}
        \begin{itemize}
            \item بالا بردن اولویت \lr{Consumer}.
            \item ارسال اشاره‌گر به داده به‌جای کپی کامل پیام.
            \item استفاده از حالت غیرمسدودکننده یا \lr{Timeout}.
            \item افزایش ظرفیت Mailbox براساس نرخ واقعی تولید و مصرف.
        \end{itemize}
    \end{solution}

        \textbf{دلایل رخ‌دادن:}
        \begin{itemize}
            \item صف مشترک باعث \lr{Head-of-Line Blocking} می‌شود و کارهای سبک پشت کارهای سنگین گیر می‌کنند.
            \item کارهای CPU-bound نخ‌ها را برای مدت زیاد اشغال می‌کنند و باعث اشباع Pool می‌شوند.
            \item کارهای I/O-bound نخ را هنگام انتظار بلوکه می‌کنند و CPU گاهی بیکار می‌ماند.
        \end{itemize}

        \textbf{راه‌حل‌ها:}
        \begin{itemize}
            \item جداکردن صف‌ها و ساختن Thread Poolهای مجزا برای انواع کارها.
            \item استفاده از \lr{Work-Stealing} برای کاهش انسداد صف.
            \item استفاده از \lr{Priority Queue} برای اجرای سریع کارهای سبک و حساس.
        \end{itemize}

    \end{solution}
